Dynamic weight 계산 공식

def calc_dynamic_weight(base, ic_observed, regime, obs_count, max_cap): ic_target = 0.020 # statistical significance threshold ic_mult = clip(ic_observed / ic_target, 0.0, 1.5) regime_mult = REGIME_WEIGHT_MULTIPLIERS[regime][signal_name] conf_factor = sqrt(obs_count / 1000) # 1000+ obs → 1.0, 100 obs → 0.32 return clip(base * ic_mult * regime_mult * conf_factor, 0.0, max_cap)
